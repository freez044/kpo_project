@startuml

skinparam classAttributeIconSize 0

class User {
    +id: int
    +name: string
    +email: string
    +password_hash: string
    +timezone: string
}

class Goal {
    +id: int
    +user_id: int
    +title: string
    +description: string
    +category: Category
    +status: GoalStatus
    +deadline: datetime
}

class Task {
    +id: int
    +user_id: int
    +goal_id: int?
    +title: string
    +description: string
    +status: TaskStatus
    +priority: TaskPriority
    +estimated_duration_minutes: int
    +deadline: datetime
    +completed_at: datetime?
}

class TaskSession {
    +id: int
    +task_id: int
    +started_at: datetime
    +ended_at: datetime?
}

class ScheduleItem {
    +id: int
    +task_id: int
    +start_at: datetime
    +end_at: datetime
}

enum Category {
    STUDY
    WORK
    SPORT
    PERSONAL
    OTHER
}

enum TaskStatus {
    BACKLOG
    PLANNED
    IN_PROGRESS
    IN_HOLD
    DONE
    CANCELLED
}

enum GoalStatus {
    ACTIVE
    IN_HOLD
    COMPLETED
    CANCELLED
}

enum TaskPriority {
    LOW
    MEDIUM
    HIGH
    URGENT
}

User "1" -- "0..*" Goal : owns
User "1" -- "0..*" Task : owns

Goal "0..1" -- "0..*" Task : contains

Task "1" -- "0..*" TaskSession : sessions
Task "1" -- "0..*" ScheduleItem : schedule

Goal ..> Category
Goal ..> GoalStatus
Task ..> TaskStatus
Task ..> TaskPriority

@enduml
